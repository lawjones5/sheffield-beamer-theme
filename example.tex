\documentclass[aspectratio=169]{beamer}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{booktabs}
\usepackage[watermark=arrows,sectionpages]{beamerthemeSheffield}

\title{Core Presentation Template}
\subtitle{A LaTeX theme for the University of Sheffield}
\author{Author Name}
\institute{The University of Sheffield}
\date{October 2026}

\begin{document}

% --- Title slide (template layout CUSTOM_2_1) -----------------------
\SheffieldTitlePage

% --- Guidance --------------------------------------------------------
\begin{frame}{Presentation guidance}
  This theme mirrors the University's core presentation template:
  \begin{itemize}
    \item Source Sans Pro for text, Source Serif Pro for display type
    \item Sheffield purple (\#440099) as the only accent colour
    \item Quiet grey canvases with watermarked divider slides
    \item Slides support the viewer --- content carries the argument
  \end{itemize}
\end{frame}

% --- Title and text (template layout BLANK) --------------------------
\begin{frame}{Title and text}
  Fusce bibendum est leo, eu bibendum justo blandit sed. Mauris sit amet
  mattis elit. Integer quis consequat lectus.

  \begin{itemize}
    \item Rutrum felis sit amet, faucibus metus
    \item Pellentesque ultricies eu turpis sit amet sollicitudin
    \item Vestibulum porttitor nulla porta eros gravida volutpat
  \end{itemize}

  Mauris sit amet mattis elit, \alert{integer quis consequat lectus}.
\end{frame}

% --- Section divider (template layout CUSTOM_3) ----------------------
